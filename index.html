function exportFilteredCSV() { const history = JSON.parse(localStorage.getItem('rdo_database') || '[]'); if (history.length === 0) return alert('Nenhum dado cadastrado para exportação.'); // Obter filtros aplicados na tela const fProj = document.getElementById('filterProjeto').value; const fData = document.getElementById('filterData').value; const fTec = document.getElementById('filterTecnico').value; const fEng = document.getElementById('filterEngenheiro').value; const fTipo = document.getElementById('filterTipoObra').value; // Filtrar os registros conforme o que o usuário selecionou const filtered = history.filter(r => { if (fProj && r.projeto !== fProj) return false; if (fData) { const parts = r.data.split('/'); const formattedRData = `${parts[2]}-${parts[1].padStart(2, '0')}-${parts[0].padStart(2, '0')}`; if (formattedRData !== fData) return false; } if (fTec && r.tecnico !== fTec) return false; if (fEng && r.engenheiro !== fEng) return false; if (fTipo && r.tipoObra !== fTipo) return false; return true; }); if (filtered.length === 0) return alert('Nenhum registro encontrado com os filtros selecionados.'); // Cabeçalho do CSV const headers = ["ID", "Data", "Projeto", "Tecnico", "Engenheiro", "Tipo de Obra", "Status", "Atividades Previstas", "Atividades Realizadas", "Assinatura"]; // Converte os dados em linhas tratadas contra aspas e quebras de linha const rows = filtered.map(r => [ r.id, `"${r.data}"`, `"${r.projeto.replace(/"/g, '""')}"`, `"${r.tecnico.replace(/"/g, '""')}"`, `"${r.engenheiro.replace(/"/g, '""')}"`, `"${r.tipoObra.replace(/"/g, '""')}"`, `"${r.status}"`, `"${(r.inicial?.atividadesPrevistas || '').replace(/"/g, '""').replace(/\n/g, ' ')}"`, `"${(r.encerramento?.atividadesRealizadas || '').replace(/"/g, '""').replace(/\n/g, ' ')}"`, `"${(r.encerramento?.assinatura || 'Sem Assinatura').replace(/"/g, '""')}"` ]); // Monta o conteúdo final utilizando Ponto e Vírgula (;) ideal para Excel em português const csvContent = [headers.join(";"), ...rows.map(row => row.join(";"))].join("\n"); // Adiciona o BOM (\uFEFF) para forçar o suporte a UTF-8 (Acentos/ç) no Excel const blob = new Blob(["\uFEFF" + csvContent], { type: "text/csv;charset=utf-8;" }); const url = URL.createObjectURL(blob); const link = document.createElement("a"); link.setAttribute("href", url); link.setAttribute("download", `rdo_banco_de_dados_${new Date().toISOString().split('T')[0]}.csv`); document.body.appendChild(link); link.click(); document.body.removeChild(link); }
